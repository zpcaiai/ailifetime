```sh
pnpm --filter @ltm/api... build
pnpm --filter @ltm/wechat-mini-core test
node apps/wechat-mini/tools/devtools-ui.mjs \
  --synthetic-ui \
  --sdk /absolute/path/to/miniprogram-automator \
```
